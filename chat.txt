async function run(cv,text){
  const ai={role:"assistant",content:""};
  cv.messages.push(ai); save(); renderList(); renderChat();
  const el=$("#chat").lastElementChild.querySelector(".body");
  const msgs=[];
  if(cfg.sys.trim())msgs.push({role:"system",content:cfg.sys});
  cv.messages.slice(0,-1).forEach(m=>{
    const o={role:m.role,content:m.content};
    if(m.images?.length)o.images=m.images.map(i=>i.split(",")[1]); // base64のみ
    msgs.push(o);
  });
  ctrl=new AbortController(); $("#send").textContent="停止"; stopSpeak(); let failed=0;
  try{
    const extra=[await webSearchFlow(text,el,ai,ctrl.signal),await calcFlow(text,el,ai,ctrl.signal)].filter(Boolean).join("\n\n"); // 履歴には保存しない
    if(extra){
      if(cfg.toolSys){if(msgs[0]?.role==="system")msgs[0].content+="\n\n"+extra;else msgs.unshift({role:"system",content:extra})} // システムプロンプトに追記
      else msgs.splice(msgs.length-1,0,{role:"system",content:extra});} // 最新のユーザー発言の直前にシステムとして挿入
    const res=await fetch(cfg.api.replace(/\/$/,"")+"/api/chat",{method:"POST",signal:ctrl.signal,
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify({model:cfg.model,messages:msgs,stream:true,think:false})});
    if(!res.ok)throw new Error(res.status+" "+await res.text());
    const rd=res.body.getReader(),dec=new TextDecoder();let buf="";
    while(true){
      const {done,value}=await rd.read(); if(done)break;
      buf+=dec.decode(value,{stream:true});
      const lines=buf.split("\n");buf=lines.pop();
      for(const ln of lines){
        if(!ln.trim())continue;
        const j=JSON.parse(ln);
        if(j.error)throw new Error(j.error);
        if(j.message?.content){ai.content+=j.message.content;el.innerHTML=md(ai.content);$("#chat").scrollTop=$("#chat").scrollHeight}
      }
    }
  }catch(e){failed=1;
    if(e.name!=="AbortError"){ai.content+=`\n\n[エラー] ${e.message}\n(Ollamaが起動しているか、OLLAMA_ORIGINS=* を設定したか確認してください)`;el.innerHTML=md(ai.content)}
  }

  if(!ai.content)cv.messages.pop();
  save();renderChat();
  if(!failed&&cfg.autoTTS&&ai.content)speak(ai.content);
}

async function speak(text,btn){
  stopSpeak();const id=spkId;text=cleanTTS(text);if(!text)return;

  const fn=ENG[cfg.ttsEngine];
  if(fn){
    try{
      let p=fn(parts[0]);p.catch(()=>{});
      for(;i<parts.length;i++){
        const url=await p;if(id!==spkId)return;
        if(i+1<parts.length){p=fn(parts[i+1]);p.catch(()=>{})} // 次を先読み

      toast(cfg.ttsEngine+" 失敗 → ブラウザ標準で読み上げます ("+(e.message||e)+")");
    }
  }
  await browserSpeak(chunks(parts.slice(i).join(""),180),id);done();
}

function calcEval(src){ // evalは使わず、安全な再帰下降パーサーで計算
  let s=src.normalize("NFKC").replace(/✖\uFE0F?|✕|×|⋅|·/g,"*").replace(/÷/g,"/").replace(/[−–—]/g,"-")
    .replace(/(?<=\d),(?=\d{3})/g,"").replace(/\s+/g,"").replace(/\^/g,"**").replace(/√\(/g,"sqrt(").replace(/√(\d+\.?\d*)/g,"sqrt($1)");
  let i=0;
  const F={sqrt:Math.sqrt,cbrt:Math.cbrt,abs:Math.abs,sin:Math.sin,cos:Math.cos,tan:Math.tan,asin:Math.asin,acos:Math.acos,atan:Math.atan,
    log:Math.log,ln:Math.log,log10:Math.log10,log2:Math.log2,exp:Math.exp,floor:Math.floor,ceil:Math.ceil,round:Math.round,pow:Math.pow,min:Math.min,max:Math.max};
  const C={pi:Math.PI,e:Math.E},err=m=>{throw new Error(m)};
  function expr(){let v=term();while(s[i]==="+"||s[i]==="-"){const o=s[i++],r=term();v=o==="+"?v+r:v-r}return v}
  function term(){let v=unary();for(;;){const c=s[i];
    if(c==="*"&&s[i+1]!=="*"){i++;v*=unary()}else if(c==="/"){i++;v/=unary()}else if(c==="%"){i++;v%=unary()}else break}return v}
  function unary(){if(s[i]==="-"){i++;return-unary()}if(s[i]==="+"){i++;return unary()}return power()}
  function power(){const b=atom();if(s[i]==="*"&&s[i+1]==="*"){i+=2;return b**unary()}return b}
  function atom(){
    if(s[i]==="("){i++;const v=expr();if(s[i++]!==")")err("括弧が閉じていません");return v}
    let m=/^(\d+\.?\d*|\.\d+)(e[+-]?\d+)?/i.exec(s.slice(i));
    if(m){i+=m[0].length;return parseFloat(m[0])}
    m=/^[a-z_][a-z0-9_]*/i.exec(s.slice(i));
    if(m){
      const id=m[0].toLowerCase();i+=m[0].length;
      if(s[i]==="("){
        i++;const a=[];
        if(s[i]!==")"){do{a.push(expr())}while(s[i]===","&&++i)}
        if(s[i++]!==")")err("括弧が閉じていません");
        if(!Object.hasOwn(F,id))err("未対応の関数: "+id);
        return F[id](...a);
      }
      if(Object.hasOwn(C,id))return C[id];
      err("未対応の記号: "+m[0]);
    }
    err("式を解釈できません: "+(s[i]??"末尾"));
  }
  const v=expr();
  if(i<s.length)err("式を解釈できません: "+s[i]);
  if(typeof v!=="number"||!isFinite(v))err("計算結果が不正です(0除算など)");
  return v;
}
const fmtNum=x=>String(+x.toPrecision(15)); // 0.1+0.2 → 0.3
// ---- 代数: 多項式を展開して同類項をまとめる / 1変数の1次・2次方程式を解く ----
// 単項式は「文字を並べて整列した文字列」をキーにする (x^2y → "xxy", 定数 → "")
const normExpr=src=>src.normalize("NFKC").replace(/✖\uFE0F?|✕|×|⋅|·/g,"*").replace(/÷/g,"/").replace(/[−–—]/g,"-")
  .replace(/(?<=\d),(?=\d{3})/g,"").replace(/\s+/g,"").replace(/\^/g,"**");
const pclean=r=>{for(const k in r)if(Math.abs(r[k])<1e-12)delete r[k];return r};
const PADD=(a,b,s=1)=>{const r={...a};for(const k in b)r[k]=(r[k]||0)+s*b[k];return pclean(r)};
const PMUL=(a,b)=>{const r={};for(const x in a)for(const y in b){const k=[...(x+y)].sort().join("");r[k]=(r[k]||0)+a[x]*b[y]}return pclean(r)};
const PCONST=p=>{const ks=Object.keys(p);return ks.length===0?0:(ks.length===1&&ks[0]===""?p[""]:null)}; // 定数でなければnull
const keyStr=k=>{const c={};for(const ch of k)c[ch]=(c[ch]||0)+1;return Object.keys(c).sort().map(ch=>ch+(c[ch]>1?"^"+c[ch]:"")).join("")};
function fmtPoly(p){
  const ks=Object.keys(p).sort((a,b)=>b.length-a.length||(a<b?-1:1));
  if(!ks.length)return"0";
  return ks.map((k,j)=>{const c=p[k],a=Math.abs(c),body=k?(a===1?"":fmtNum(a))+keyStr(k):fmtNum(a);
    return(j===0?(c<0?"-":""):(c<0?" - ":" + "))+body}).join("");
}
function fmtFrac(x){
  if(Number.isInteger(+x.toPrecision(12)))return fmtNum(x);
  for(let q=2;q<=1000;q++){const p=Math.round(x*q);if(Math.abs(x-p/q)<1e-9)return`${p}/${q} (≈${fmtNum(x)})`}
  return fmtNum(x);
}
function algEval(src){
  const s=normExpr(src),err=m=>{throw new Error(m)};
  let i=0,depth=0;const contrib={};
  const num=n=>pclean({"":n}),FN=/^(sqrt|cbrt|abs|sin|cos|tan|asin|acos|atan|log10|log2|log|ln|exp|floor|ceil|round|pow|min|max)(?=\()/i;
  const rec=(top,t,sg)=>{if(top)for(const k in t)(contrib[k]??=[]).push(sg*t[k])}; // 最上位の項ごとの係数を記録
  function expr(){
    const top=depth++===0;let t=term();rec(top,t,1);let v=t;
    while(s[i]==="+"||s[i]==="-"){const sg=s[i++]==="+"?1:-1;t=term();rec(top,t,sg);v=PADD(v,t,sg)}
    depth--;return v;
  }
  function term(){
    let v=unary();
    for(;;){const c=s[i];
      if(c==="*"&&s[i+1]!=="*"){i++;v=PMUL(v,unary())}
      else if(c==="/"){i++;const d=PCONST(unary());if(!d)err("0または変数での割り算には未対応");v=PMUL(v,num(1/d))}
      else if(c&&/[a-z(]/i.test(c)){v=PMUL(v,power())} // 暗黙の掛け算: 8x, 2(x+1), (x+1)(x-1)
      else break}
    return v;
  }
  function unary(){if(s[i]==="-"){i++;return PMUL(unary(),num(-1))}if(s[i]==="+"){i++;return unary()}return power()}
  function power(){
    const b=atom();
    if(s[i]==="*"&&s[i+1]==="*"){
      i+=2;const e=PCONST(unary());if(e===null)err("指数に変数は未対応");
      const bc=PCONST(b);if(bc!==null)return num(Math.pow(bc,e));
      if(!Number.isInteger(e)||e<0||e>8)err("変数を含む式の指数は0〜8の整数のみ対応");
      let r=num(1);for(let j=0;j<e;j++)r=PMUL(r,b);return r;
    }
    return b;
  }
  function atom(){
    if(s[i]==="("){i++;const v=expr();if(s[i++]!==")")err("括弧が閉じていません");return v}
    let m=/^(\d+\.?\d*|\.\d+)(e[+-]?\d+)?/i.exec(s.slice(i));
    if(m){i+=m[0].length;return num(parseFloat(m[0]))}
    m=FN.exec(s.slice(i));
    if(m){
      const f=m[1].toLowerCase();i+=m[0].length+1;const a=[];
      if(s[i]!==")"){do{const c=PCONST(expr());if(c===null)err("関数の引数に変数は未対応");a.push(c)}while(s[i]===","&&++i)}
      if(s[i++]!==")")err("括弧が閉じていません");
      return num(Math[f==="ln"?"log":f](...a));
    }
    m=/^[a-z]/i.exec(s.slice(i));
    if(m){i++;return{[m[0].toLowerCase()]:1}}
    err("式を解釈できません: "+(s[i]??"末尾"));
  }
  const p=expr();
  if(i<s.length)err("式を解釈できません: "+s[i]);
  return{p,contrib};
}
function algSolve(src){
  const parts=src.normalize("NFKC").split("=").map(x=>x.trim());
  if(parts.length>2)throw new Error("「=」が複数あります");
  const A=algEval(parts[0]||"0"),eq=parts.length===2&&parts[1]!=="";
  const nt=(...a)=>a.filter(Boolean).join(" / ");
  // 同類項の計算の途中経過 (例: xの係数: 8 + 9 = 17)
  const steps=Object.entries(A.contrib).filter(([,v])=>v.length>1).map(([k,v])=>
    `${k?keyStr(k)+"の係数":"定数項"}: ${v.map((n,j)=>j?(n<0?` + (${fmtNum(n)})`:` + ${fmtNum(n)}`):fmtNum(n)).join("")} = ${fmtNum(v.reduce((a,b)=>a+b,0))}`).join(" / ");
  if(!eq)return{result:fmtPoly(A.p),note:steps};
  const B=algEval(parts[1]),D=PADD(A.p,B.p,-1);
  const vars=[...new Set(Object.keys(D).join("").split(""))].filter(Boolean);
  if(!vars.length)return{result:Object.keys(D).length===0?"恒等的に成立 (常に真)":`成立しない (偽): ${fmtPoly(A.p)} ≠ ${fmtPoly(B.p)}`,note:""};
  if(vars.length>1)return{result:`整理: ${fmtPoly(D)} = 0 (変数が複数のため解は求めません)`,note:steps};
  const v=vars[0],c0=D[""]||0,c1=D[v]||0,c2=D[v+v]||0,deg=Math.max(...Object.keys(D).map(k=>k.length));
  if(deg===1)return{result:`${v} = ${fmtFrac(-c0/c1)}`,note:nt(steps,`移項して ${fmtPoly(D)} = 0`)};
  if(deg===2){
    const d=c1*c1-4*c2*c0,note=nt(steps,`整理して ${fmtPoly(D)} = 0, 判別式 D = ${fmtNum(d)}`);
    if(d<0)return{result:`実数解なし (複素数解: ${v} = ${fmtNum(-c1/(2*c2))} ± ${fmtNum(Math.sqrt(-d)/Math.abs(2*c2))}i)`,note};
    const r1=(-c1+Math.sqrt(d))/(2*c2),r2=(-c1-Math.sqrt(d))/(2*c2);
    return{result:Math.abs(d)<1e-12?`${v} = ${fmtFrac(r1)} (重解)`:`${v} = ${fmtFrac(r1)}, ${v} = ${fmtFrac(r2)}`,note};
  }
  return{result:`整理: ${fmtPoly(D)} = 0 (3次以上の方程式は未対応)`,note:steps};
}
async function calcFlow(text,el,ai,sig){
  if(!cfg.autoCalc||!text)return"";
  try{
    el.textContent="🧮 計算の要否を判断中…";
    const mode=(await llmJSON(CALC_JUDGE,text,sig)).mode;
    if(mode!=="calc"&&mode!=="algebra")return"";
    el.textContent="🧮 式を作成中…";
    const ex=(await llmJSON(mode==="algebra"?ALG_EXPR:CALC_EXPR,text,sig)).exprs;
    const list=(Array.isArray(ex)?ex:[ex]).filter(x=>typeof x==="string"&&x.trim()).slice(0,5);
    if(!list.length)return"";
    el.textContent="🧮 計算中…";
    ai.calc=list.map(x=>{try{
      if(mode==="algebra"){const a=algSolve(x);return{expr:x,result:a.result,note:a.note,alg:1}}
      return{expr:x,result:fmtNum(calcEval(x))};
    }catch(e){return{expr:x,error:e.message}}});
  }catch(e){if(e.name==="AbortError")throw e;return""}
  const ok=ai.calc.filter(c=>c.result!==undefined);
  if(!ok.length)return"";
  return"以下は、あなた自身が電卓(JavaScript)で正確に計算・整理した結果です(代数式は同類項をまとめて整理済み)。ユーザーが提示した資料ではありません。「ご提示の計算結果」のような言い方は避け、この値を信頼してそのまま回答に使ってください(自分で再計算して別の値にしないこと)。\n"+ok.map(c=>`${c.expr}${c.alg?"  →  ":" = "}${c.result}`+(c.note?`\n  (途中計算: ${c.note})`:"")).join("\n");
}
