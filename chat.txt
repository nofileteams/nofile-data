<script>
const $=s=>document.querySelector(s);
const DEF={api:"http://localhost:11434",model:"liminal-space/dreamcore-ai:latest",title:"AAA",theme:"neon",color:"#ff1a1a",sys:"",userName:"YOU",aiName:"AI",bgMode:"default",bgColor:"#000000",bgImage:""};
const load=(k,d)=>{try{return JSON.parse(localStorage.getItem(k))??d}catch{return d}};
let cfg={...DEF,...load("oc_cfg",{})}, convs=load("oc_convs",[]), cur=null, atts=[], ctrl=null;

function save(){
  try{localStorage.setItem("oc_convs",JSON.stringify(convs))}
  catch{ // 容量超過時は画像を除外して保存
    try{localStorage.setItem("oc_convs",JSON.stringify(convs.map(c=>({...c,messages:c.messages.map(m=>({...m,images:undefined}))}))))}catch{}
  }
}
const saveCfg=()=>{try{localStorage.setItem("oc_cfg",JSON.stringify(cfg))}catch{}};

function applyCfg(){
  document.body.classList.toggle("liminal",cfg.theme==="liminal");
  document.documentElement.style.setProperty("--c",cfg.color);
  const useImg=cfg.bgMode==="image"&&cfg.bgImage;
  document.body.classList.toggle("bgimg",!!useImg);
  document.body.classList.toggle("custombg",!!useImg||cfg.bgMode==="color");
  document.body.style.background=useImg?`url(${cfg.bgImage}) center/cover fixed`:cfg.bgMode==="color"?cfg.bgColor:"";
  $("#title").textContent=cfg.title; $("#side-title").textContent=cfg.title; document.title=cfg.title;
  $("#model-tag").textContent=cfg.model;
}
const esc=s=>s.replace(/[&<>]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;"}[c]));
function md(t){
  let h=esc(t);
  h=h.replace(/```(\w*)\n?([\s\S]*?)(```|$)/g,(_,l,c)=>`<pre>${c}</pre>`);
  h=h.replace(/`([^`\n]+)`/g,"<code>$1</code>").replace(/\*\*([^*\n]+)\*\*/g,"<b>$1</b>");
  return h;
}
const curConv=()=>convs.find(c=>c.id===cur);

function renderList(){
  const l=$("#list"); l.innerHTML="";
  convs.forEach(c=>{
    const d=document.createElement("div"); d.className="item"+(c.id===cur?" on":"");
    d.innerHTML=`<span>${esc(c.title)}</span><b title="削除">✕</b>`;
    d.onclick=()=>{cur=c.id;renderList();renderChat()};
    d.querySelector("b").onclick=e=>{e.stopPropagation();convs=convs.filter(x=>x.id!==c.id);if(cur===c.id)cur=null;save();renderList();renderChat()};
    l.appendChild(d);
  });
}
function msgEl(m){
  const d=document.createElement("div"); d.className="msg "+m.role;
  let h=`<div class="role"><span>${esc(m.role==="user"?cfg.userName:cfg.aiName)}</span><button class="cp">コピー</button></div>`;
  (m.images||[]).forEach(i=>h+=`<img src="${i}">`);
  (m.files||[]).forEach(f=>h+=`<span class="chip">📄 ${esc(f)}</span>`);
  h+=`<div class="body">${md(m.shown??m.content)}</div>`;
  d.innerHTML=h;
  const b=d.querySelector(".cp");
  b.onclick=async()=>{
    const t=m.shown??m.content; // 添付ファイル本文は含めず、入力した文字のみ
    try{await navigator.clipboard.writeText(t)}
    catch{const a=document.createElement("textarea");a.value=t;document.body.appendChild(a);a.select();document.execCommand("copy");a.remove()}
    b.textContent="コピーしました";setTimeout(()=>b.textContent="コピー",1500);
  };
  return d;
}
function renderChat(){
  const c=$("#chat"); c.innerHTML="";
  const cv=curConv();
  if(!cv||!cv.messages.length){c.innerHTML='<div id="empty">. . .</div>';return}
  cv.messages.forEach(m=>c.appendChild(msgEl(m)));
  c.scrollTop=c.scrollHeight;
}
function renderAtts(){
  const a=$("#atts"); a.innerHTML="";
  atts.forEach((x,i)=>{
    const d=document.createElement("div"); d.className="a";
    d.innerHTML=(x.type==="image"?`<img src="${x.data}">`:`📄 ${esc(x.name)}`)+`<i>✕</i>`;
    d.querySelector("i").onclick=()=>{atts.splice(i,1);renderAtts()};
    a.appendChild(d);
  });
}
function addFiles(files){
  [...files].forEach(f=>{
    const r=new FileReader();
    if(f.type.startsWith("image/")){
      r.onload=()=>{atts.push({type:"image",name:f.name,data:r.result});renderAtts()};
      r.readAsDataURL(f);
    }else{
      r.onload=()=>{atts.push({type:"text",name:f.name,data:r.result});renderAtts()};
      r.readAsText(f);
    }
  });
}

async function send(){
  if(ctrl){ctrl.abort();return}
  const text=$("#in").value.trim();
  if(!text&&!atts.length)return;
  if(!curConv()){const c={id:Date.now(),title:(text||atts[0].name).slice(0,28),messages:[]};convs.unshift(c);cur=c.id}
  const cv=curConv();
  const imgs=atts.filter(a=>a.type==="image").map(a=>a.data);
  const tfs=atts.filter(a=>a.type==="text");
  let content=text;
  tfs.forEach(f=>content+=`\n\n[添付ファイル: ${f.name}]\n\`\`\`\n${f.data}\n\`\`\``);
  cv.messages.push({role:"user",content,shown:text,images:imgs.length?imgs:undefined,files:tfs.map(f=>f.name)});
  $("#in").value="";$("#in").style.height="auto";atts=[];renderAtts();
  const ai={role:"assistant",content:""};
  cv.messages.push(ai); save(); renderList(); renderChat();
  const el=$("#chat").lastElementChild.querySelector(".body");
  const msgs=[];
  if(cfg.sys.trim())msgs.push({role:"system",content:cfg.sys});
  cv.messages.slice(0,-1).forEach(m=>{
    const o={role:m.role,content:m.content};
    if(m.images?.length)o.images=m.images.map(i=>i.split(",")[1]); // base64のみ
    msgs.push(o);
  });
  ctrl=new AbortController(); $("#send").textContent="停止";
  try{
    const res=await fetch(cfg.api.replace(/\/$/,"")+"/api/chat",{method:"POST",signal:ctrl.signal,
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify({model:cfg.model,messages:msgs,stream:true,think:false})});
    if(!res.ok)throw new Error(res.status+" "+await res.text());
    const rd=res.body.getReader(),dec=new TextDecoder();let buf="";
    while(true){
      const {done,value}=await rd.read(); if(done)break;
      buf+=dec.decode(value,{stream:true});
      const lines=buf.split("\n");buf=lines.pop();
      for(const ln of lines){
        if(!ln.trim())continue;
        const j=JSON.parse(ln);
        if(j.error)throw new Error(j.error);
        if(j.message?.content){ai.content+=j.message.content;el.innerHTML=md(ai.content);$("#chat").scrollTop=$("#chat").scrollHeight}
      }
    }
  }catch(e){
    if(e.name!=="AbortError"){ai.content+=`\n\n[エラー] ${e.message}\n(Ollamaが起動しているか、OLLAMA_ORIGINS=* を設定したか確認してください)`;el.innerHTML=md(ai.content)}
  }
  ctrl=null;$("#send").textContent="送信";
  if(!ai.content)cv.messages.pop();
  save();renderChat();
}

// events
$("#send").onclick=send;
$("#in").addEventListener("keydown",e=>{if(e.key==="Enter"&&!e.shiftKey&&!e.isComposing){e.preventDefault();send()}});
$("#in").addEventListener("input",e=>{e.target.style.height="auto";e.target.style.height=Math.min(e.target.scrollHeight,200)+"px"});
$("#in").addEventListener("paste",e=>{const f=[...e.clipboardData.files];if(f.length){e.preventDefault();addFiles(f)}});
$("#attach").onclick=()=>$("#file").click();
$("#file").onchange=e=>{addFiles(e.target.files);e.target.value=""};
$("#new").onclick=()=>{cur=null;renderList();renderChat()};
$("#tog").onclick=()=>$("#side").classList.toggle("hide");
$("#clearall").onclick=()=>{if(confirm("全ての会話履歴を削除しますか？")){convs=[];cur=null;save();renderList();renderChat()}};
let dc=0;
addEventListener("dragenter",e=>{e.preventDefault();dc++;$("#drop").style.display="flex"});
addEventListener("dragleave",()=>{if(--dc<=0){dc=0;$("#drop").style.display="none"}});
addEventListener("dragover",e=>e.preventDefault());
addEventListener("drop",e=>{e.preventDefault();dc=0;$("#drop").style.display="none";addFiles(e.dataTransfer.files)});

// settings
let tmpBg="";
$("#s-bgf").onchange=e=>{
  const f=e.target.files[0];if(!f)return;
  const img=new Image(),u=URL.createObjectURL(f);
  img.onload=()=>{ // 縮小してJPEG化 (localStorage容量対策)
    const s=Math.min(1,1920/Math.max(img.width,img.height)),cv=document.createElement("canvas");
    cv.width=img.width*s;cv.height=img.height*s;cv.getContext("2d").drawImage(img,0,0,cv.width,cv.height);
    tmpBg=cv.toDataURL("image/jpeg",.8);URL.revokeObjectURL(u);$("#s-bg").value="image";
  };
  img.src=u;
};
async function openSet(){
  $("#s-api").value=cfg.api;$("#s-model").value=cfg.model;$("#s-title").value=cfg.title;
  $("#s-theme").value=cfg.theme;$("#s-color").value=cfg.color;$("#s-sys").value=cfg.sys;
  $("#s-un").value=cfg.userName;$("#s-an").value=cfg.aiName;
  $("#s-bg").value=cfg.bgMode;$("#s-bgc").value=cfg.bgColor;tmpBg=cfg.bgImage;
  $("#modal").classList.add("show");
  try{const r=await fetch(cfg.api.replace(/\/$/,"")+"/api/tags");const j=await r.json();
    $("#models").innerHTML=j.models.map(m=>`<option value="${esc(m.name)}">`).join("")}catch{}
}
function closeSet(){
  cfg={api:$("#s-api").value.trim()||DEF.api,model:$("#s-model").value.trim()||DEF.model,title:$("#s-title").value||DEF.title,
    theme:$("#s-theme").value,color:$("#s-color").value,sys:$("#s-sys").value,
    userName:$("#s-un").value.trim()||DEF.userName,aiName:$("#s-an").value.trim()||DEF.aiName,
    bgMode:$("#s-bg").value,bgColor:$("#s-bgc").value,bgImage:tmpBg};
  saveCfg();applyCfg();if(!ctrl)renderChat();$("#modal").classList.remove("show");
}
$("#set").onclick=openSet;$("#s-close").onclick=closeSet;
$("#s-reset").onclick=()=>{cfg={...DEF};saveCfg();applyCfg();openSet()};
$("#modal").onclick=e=>{if(e.target.id==="modal")closeSet()};

applyCfg();renderList();renderChat();
</script>
