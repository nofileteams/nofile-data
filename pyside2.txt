import sys
from PySide6.QtCore import Qt, QUrl, QTimer, QObject, Slot, QPoint
from PySide6.QtWebChannel import QWebChannel
from PySide6.QtWidgets import QApplication, QWidget, QLabel, QLineEdit, QTextEdit, QPushButton, QFrame, QVBoxLayout, QMainWindow, QScrollArea
from PySide6.QtWebEngineWidgets import QWebEngineView
from PySide6.QtGui import QIcon, QPixmap, QPainter, QTransform


class test(QFrame):
	def __init__(self, parent=None):
		super().__init__(parent)
		self.point = QPoint()
	def mousePressEvent(self, event):
		if event.button() == Qt.MouseButton.LeftButton:
			self.point = event.globalPosition().toPoint() - self.frameGeometry().topLeft()
	def mouseMoveEvent(self, event):
		if event.buttons() == Qt.MouseButton.LeftButton:
			self.move(event.globalPosition().toPoint() - self.point)


class Windows(QMainWindow):
	def __init__(self):
		super().__init__()
		self.setWindowTitle("test")
		self.resize(1000, 700)
		self.setFixedSize(self.size())
		self.setStyleSheet("background-color: black;")
		self.box = test(self)
		self.box.setStyleSheet("background-color: red;")


app = QApplication(sys.argv)
main = Windows()
main.show()
sys.exit(app.exec())

import sys
from PySide6.QtCore import Qt, QUrl, QTimer, QObject, Slot, QPoint
from PySide6.QtWebChannel import QWebChannel
from PySide6.QtWidgets import QApplication, QWidget, QLabel, QLineEdit, QTextEdit, QPushButton, QFrame, QVBoxLayout, QMainWindow, QScrollArea
from PySide6.QtWebEngineWidgets import QWebEngineView
from PySide6.QtGui import QIcon, QPixmap, QPainter, QTransform

class windows(QMainWindow):
	def __init__(self):
		super().__init__()
		self.setWindowTitle("Youtube")
		self.resize(1000, 700)
		self.web = QWebEngineView(self)
		self.web.setGeometry(self.rect())
		self.web.setUrl(QUrl("https://youtube.com"))
		
		self.fullscreen = False
		
		self.timer = QTimer(self)
		self.timer.timeout.connect(lambda: self.web.setGeometry(self.rect()))
		self.timer.start(10)
	def keyPressEvent(self, event):
		if event.key() == Qt.Key_F11:
			if self.fullscreen == False:
				self.showFullScreen()
				self.fullscreen = True
			else:
				self.showNormal()
				#self.showMaximized()
				self.fullscreen = False
		


app = QApplication(sys.argv)
main = windows()
main.show()
sys.exit(app.exec())
